\begin{abstract}
We describe a generative framework for data whose \emph{shape} is itself random: a discrete structure (a set, a sequence, a graph, a tree) whose elements carry values in arbitrary state spaces. The structure is generated by a Markov jump process on a global clock; every value is generated by its own denoising Markov process on its own clock, started from the reference distribution at the moment its site is created. Because a site is born at clock zero, the only learned objects are the reverse rates of the structure process and the reverse generators of the site processes: no transition kernel between a noisy structure and noisy content is ever modelled. The construction is a product of independent noising processes given the data, so the multi-time decomposition of \emph{Diffuse Everything} applies verbatim and licenses any monotone clock policy at sampling time. Ordinary multimodal diffusion, masked diffusion, Edit Flows and OneFlow are instances.
\end{abstract}

\section{Objects}

\paragraph{Structure space.}
Let $\Sigmasp$ be a countable set of \emph{skeletons}. A skeleton $\sigma\in\Sigmasp$ indexes a finite collection of \emph{sites} $\sites(\sigma)$. Examples: $\Sigmasp=\N$ with $\sites(n)=\{1,\dots,n\}$ unordered (sets); strings with $\sites(\sigma)$ the positions (sequences); trees or graphs with $\sites(\sigma)$ the nodes and edges; a singleton $\Sigmasp$ (a fixed skeleton), which recovers ordinary multimodal diffusion.

\paragraph{Site state spaces.}
Every site $a$ has a \emph{type} with a state space $\M_a$ (continuous, discrete, or otherwise) equipped with a \emph{noising process}: a Markov process on $\M_a$ with generator $\gen^a_s$ in its own time $s\in[0,1]$, whose law at the fully noised end is a fixed reference measure $\pi_a$, reached \emph{exactly}. In generation time we write $s=0$ for the reference end and $s=1$ for the data end. Instances: the Gaussian flow-matching path $x_s=s\,x_1+(1-s)\,\varepsilon$ with $\pi_a=\mathcal N(0,I)$; an absorbing (masked) CTMC with $\pi_a=\delta_{\mathrm{MASK}}$; any finite-time flow-matching path on a manifold. A \emph{degenerate} site type has no noising process: its value is fixed at birth (its clock jumps from $0$ to $1$).

\paragraph{Data space.}
\begin{equation}
\Z \;=\; \bigsqcup_{\sigma\in\Sigmasp}\;\prod_{a\in\sites(\sigma)}\M_a ,
\end{equation}
a bundle over $\Sigmasp$. A datum is $(\sigma_1,(m_a)_{a\in\sites(\sigma_1)})$.

\paragraph{Times.}
One global clock $t\in[0,1]$ for the structure and one clock $s_a\in[0,1]$ per site. For a state with skeleton $\sigma$ the time domain is $[0,1]\times[0,1]^{\sites(\sigma)}$. The \emph{augmented state} is
\begin{equation}
z=\bigl(\sigma,\,(m_a,s_a)_{a\in\sites(\sigma)}\bigr).
\end{equation}

\section{Reference construction: a product given the data}
\label{sec:reference}

Fix a datum $(\sigma_1,m)$. In noising direction:
\begin{enumerate}
\item \textbf{Structure noising.} A Markov process on $\Sigmasp$ started at $\sigma_1$ with generator $\gen^{\Sigmasp}$: deletion of sites in the simplest case, but in general any CTMC on $\Sigmasp$ (deletions, merges, type changes). Its law at global time $t$ is $K_t(\sigma_1,\cdot)$. When a site leaves the skeleton its value is discarded (marginalised).
\item \textbf{Site noising.} Independently, every site $a\in\sites(\sigma_1)$ runs its own noising process on its own clock; its law at clock $s_a$ is $k^a_{s_a}(m_a,\cdot)$.
\end{enumerate}
The joint marginal at times $(t,\{s_a\})$ is therefore
\begin{equation}
\label{eq:product}
p_{t,s}\bigl(\sigma,(m_a)_{a\in\sites(\sigma)}\,\big|\,\sigma_1,m\bigr)
\;=\;K_t(\sigma_1,\sigma)\;\prod_{a\in\sites(\sigma)}k^a_{s_a}(m_a,\cdot):
\end{equation}
a product over ``modalities'' whose \emph{number and identity are random and determined by the structure component}. This is the only structural assumption, and it is exactly what the multi-time decomposition of Diffuse Everything requires.

\section{Generative process}

\begin{definition}[Clock policy]
A clock policy assigns to every site a non-decreasing clock $s_a(t)$ with $s_a=0$ at the moment the site is created and $s_a=1$ by the end of generation (the horizon may exceed $t=1$). Write $\rho_a=\dd s_a/\dd t\ge 0$.
\end{definition}

The generative process is the Markov process on the augmented state space with generator
\begin{equation}
\label{eq:generator}
\begin{aligned}
\rgen_t f(z)\;=\;&\sum_{\sigma'}q_t(\sigma\to\sigma'\mid z)\int\Bigl[f\bigl(\sigma',\,m\cup m_{\mathrm{new}},\,s\cup 0\bigr)-f(z)\Bigr]\prod_{a\in\sites(\sigma')\setminus\sites(\sigma)}\pi_a(\dd m_a)
&&\text{(structure)}\\
&+\sum_{a\in\sites(\sigma)}\rho_a(t)\,\rgen^{a}_{s_a}[z]\,f(z)
&&\text{(sites)}
\end{aligned}
\end{equation}
where
\begin{itemize}
\item $q_t(\sigma\to\sigma'\mid z)$ are the \emph{reverse structure rates}: the time reversal of $\gen^{\Sigmasp}$, conditioned on the whole current state (skeleton, all site values at their own clocks, all clocks). Sites created by the transition are born with value drawn from $\pi_a$ and clock $0$; sites removed by it are dropped;
\item $\rgen^a_s[z]$ is the \emph{reverse site generator}: the standard time reversal of the site's noising process (drift plus score for diffusions, velocity field for flows, reverse rate matrix for CTMCs) evaluated with the joint marginal, i.e.\ conditioned on $z$; it also transports the clock, $\partial_{s_a}$.
\end{itemize}

\begin{theorem}[Reversal, informal]
\label{thm:reversal}
Assume the reference construction of Section~\ref{sec:reference}. For any clock policy with births at clock zero, the time reversal of the joint process has generator~\eqref{eq:generator}, with $q_t$ and $\rgen^a$ given by the single-component reversal formulas evaluated with the joint marginal~\eqref{eq:product}.
\end{theorem}

\begin{proof}[Proof sketch]
Apply the multi-time reversal decomposition of Diffuse Everything to the augmented state. The structure component is a CTMC reversal with the joint marginal in its ratio; each site component is a single-modality reversal on its own clock. The birth kernel factorises as (structure jump)\,$\otimes\,\pi_a$ because the marginal of a site at clock $0$ is $\pi_a$, independent of everything else.
\end{proof}

\begin{remark}[Why births happen at clock zero]
A policy that creates a site at $s_a>0$ needs, inside the structure transition, a learned kernel $p(m_a\mid z)$ at that noise level: a model of noisy content conditioned on a noisy structure. Birth at clock zero makes the only learned objects the reverse structure rates and the per-site reverse generators; no transition kernel is ever modelled. The alternative (birth at $s_a=t$ with a learned conditional) is what a ``reveal head'' does, and it introduces a train/test mismatch whenever that conditional is approximate.
\end{remark}

\section{Training}

Sample a datum, $t\sim U(0,1)$ and $s_a\sim U(0,1)$ independently for all sites (the product measure on the time cube), build $z$ from Section~\ref{sec:reference}, and minimise
\begin{equation}
\label{eq:loss}
\E\Bigl[\ell^{\Sigmasp}\bigl(q_\theta(\cdot\mid z),\ \text{structure targets}\bigr)\Bigr]
\;+\;\E\Bigl[\sum_{a\in\sites(\sigma)}\ell^{a}\bigl(\rgen^{a}_\theta[z],\ \text{site targets at }s_a\bigr)\Bigr],
\end{equation}
where $\ell^{\Sigmasp}$ is the loss of the structure reversal (a Bregman divergence for rates, cross-entropy for the types of created sites) and $\ell^a$ is the standard loss of the site's process (denoising score matching, flow matching, or the masked-diffusion cross-entropy), each evaluated at the joint state.

\begin{remark}[Coupled versus decoupled clocks]
Training on the product measure is what licenses \emph{every} clock policy at sampling time. Training instead on the push-forward of one policy, $s_a=\varphi(t,\tau_a)$ with $\tau_a$ the birth time of $a$, places every training state on the surface swept by that policy and licenses that policy only. The two choices define the same model class and differ only in the law of the training times.
\end{remark}

\section{Design axes}

A new instance of the framework has to choose:
\begin{enumerate}
\item \textbf{$\Sigmasp$ and its noising CTMC.} Which structure is grown and by which edits: counts for sets; positioned deletions for sequences; node and edge deletions for graphs; subtree deletions for hierarchies.
\item \textbf{Site types.} Which parts are degenerate (born clean, handled by the structure loss) and which carry a noising process on their own clock.
\item \textbf{Clock policy.} Synchronised to a common horizon, unit-rate with a tail, phase-separated (``grow, then denoise''), or per-site adaptive. Under product-measure training this is a sampling-time choice.
\item \textbf{Granularity of clocks.} Per site, per group of coordinates, or per coordinate; finer clocks make more conditioning tasks native time paths (a given coordinate is a coordinate held at clock $1$).
\item \textbf{Structure as a process versus structure as a value.} The alternative is a fixed maximal skeleton with ``absent'' encoded as a site value (padding). The framework contains it ($\Sigmasp$ a singleton, absence a symbol) and makes explicit what is lost: the structure reversal is then carried out by the site processes, which must model absence jointly with content.
\end{enumerate}

Permutation invariance is not part of the framework; it enters only through the choice $\Sigmasp=\N$ together with an equivariant network. For sequences one takes $\Sigmasp$ to be strings with positioned edits (the reverse process of Edit Flows) and the same site machinery; for graphs, node and edge sites with their own clocks.

\begin{example}[Layouts]
$\Sigmasp=\N$ with unordered sites; a site carries a degenerate class $y\in\mathcal Y$ and a Gaussian-path box $x\in\R^4$ with one clock per box. The structure noising process deletes each element at an i.i.d.\ time with $\Pr(\tau\le t)=\kappa(t)$; its exact reversal is a birth process with intensity $\kappa'(t)/(1-\kappa(t))\cdot\E[\,n-n_t\mid z\,]$ and class distribution $p(y\mid z)$, both learned. The box drift is the flow-matching velocity $v_\theta(z;a)\approx\E[x^1_a-\varepsilon_a\mid z]$ on the element's own clock. Synchronised clocks use $s_a=(t-\tau_a)/(1-\tau_a)$; unit-rate clocks use $s_a=\min(t-\tau_a,1)$ with horizon $1+t_{\max}$; the phase-separated policy freezes every clock until $\kappa=1$.
\end{example}
